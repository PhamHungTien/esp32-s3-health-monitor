\section{Mục tiêu công việc}

Chức năng té ngã cần phân biệt va chạm thông thường với chuỗi rơi--va đập--bất động. Phần GPS phải nhận liên tục dữ liệu NMEA, kiểm tra checksum, phân biệt các trạng thái vận hành và công bố dữ liệu nhất quán cho OLED, Serial và dashboard web.

\section{Mô-đun IMU, đếm bước và té ngã}

\subsection{Khởi tạo và chuẩn hóa}

Thiết bị I2C tại 0x68 trả về \Code{WHO\_AM\_I=0x74}. Em đọc sáu byte dữ liệu của các trục $x,y,z$, đổi sang đơn vị $g$ và tính độ lớn:

\begin{center}
\small
\begin{tabularx}{0.90\textwidth}{C{2.1cm} C{2.0cm} Y}
\toprule
\textbf{Thanh ghi} & \textbf{Giá trị} & \textbf{Mục đích}\\
\midrule
0x75 & chỉ đọc & Kiểm tra mã \Code{WHO\_AM\_I}\\
0x6B & 0x00 & Đưa IMU ra khỏi chế độ ngủ\\
0x1C & 0x00 & Chọn thang gia tốc $\pm 2g$\\
0x1A & 0x03 & Bật bộ lọc số mức 3\\
0x3B & đọc 6 byte & Lấy liên tiếp $a_x,a_y,a_z$\\
\bottomrule
\end{tabularx}
\normalsize
\end{center}

\[
 A=\sqrt{a_x^2+a_y^2+a_z^2}.
\]

Ở trạng thái đặt yên, phép đo thực tế đạt khoảng 1,07--1,08$g$, phù hợp với gia tốc trọng trường và được dùng làm mốc bất động.

\subsection{Máy trạng thái phát hiện té ngã}

Em triển khai bốn trạng thái để kiểm tra đầy đủ diễn biến của một lần té ngã:

\begin{enumerate}
  \item \textbf{NORMAL:} theo dõi bình thường;
  \item \textbf{FREEFALL:} xác nhận gia tốc thấp trong một khoảng ngắn;
  \item \textbf{IMPACT:} chờ va đập mạnh trong cửa sổ thời gian giới hạn;
  \item \textbf{ALERT:} chỉ kích hoạt sau khi có giai đoạn ít chuyển động sau va đập.
\end{enumerate}

Mọi trạng thái trung gian đều có timeout để quay về NORMAL nếu chuỗi sự kiện không hoàn chỉnh. Cấu trúc này loại nhiều báo giả do vung tay hoặc đặt thiết bị mạnh xuống bàn.

\Needspace{0.42\textheight}
\thispagestyle{fancy}
\begin{lstlisting}[style=firmware,caption={Các điều kiện quyết định trong máy trạng thái té ngã}]
(*@\VNCodeComment{Bắt đầu nhánh xử lý khi máy trạng thái đang ở điều kiện vận động bình thường.}@*)
case FALL_NORMAL:
(*@\VNCodeComment{Kiểm tra tổng gia tốc nhỏ hơn 0,45 g, dấu hiệu cơ thể có thể đang rơi tự do.}@*)
  if (accelerationG < 0.45f) {
(*@\VNCodeComment{Lưu mốc bắt đầu mức gia tốc thấp nếu đây là mẫu thấp đầu tiên.}@*)
    if (lowGStartMs == 0) lowGStartMs = now;
(*@\VNCodeComment{Yêu cầu điều kiện gia tốc thấp kéo dài ít nhất 120 ms để loại nhiễu tức thời.}@*)
    if (now - lowGStartMs >= 120) {
(*@\VNCodeComment{Chuyển máy trạng thái sang giai đoạn rơi tự do đã được xác nhận.}@*)
      fallState = FALL_FREE_FALL;
(*@\VNCodeComment{Lưu thời điểm chuyển trạng thái để kiểm soát thời hạn của giai đoạn tiếp theo.}@*)
      fallStateMs = now;
(*@\VNCodeComment{Kết thúc khối xác nhận đủ thời gian gia tốc thấp.}@*)
    }
(*@\VNCodeComment{Đi vào nhánh này khi tổng gia tốc đã trở lại từ 0,45 g trở lên.}@*)
  } else {
(*@\VNCodeComment{Xóa mốc gia tốc thấp vì chuỗi rơi tự do không còn liên tục.}@*)
    lowGStartMs = 0;
(*@\VNCodeComment{Kết thúc phép kiểm tra rơi tự do.}@*)
  }
(*@\VNCodeComment{Kiểm tra xung va đập trực tiếp lớn hơn 2,8 g.}@*)
  if (accelerationG > 2.8f) {
(*@\VNCodeComment{Chuyển sang trạng thái va đập khi phát hiện xung mạnh.}@*)
    fallState = FALL_IMPACT;
(*@\VNCodeComment{Lưu thời điểm va đập để bắt đầu cửa sổ kiểm tra bất động.}@*)
    fallStateMs = now;
(*@\VNCodeComment{Kết thúc nhánh phát hiện va đập trực tiếp.}@*)
  }
(*@\VNCodeComment{Thoát khỏi nhánh trạng thái bình thường của câu lệnh switch.}@*)
  break;
(*@\VNCodeComment{Bắt đầu nhánh chờ va đập sau khi đã xác nhận rơi tự do.}@*)
case FALL_FREE_FALL:
(*@\VNCodeComment{Xác nhận va đập nếu tổng gia tốc vượt 2,2 g trong cửa sổ rơi tự do.}@*)
  if (accelerationG > 2.2f) {
(*@\VNCodeComment{Chuyển sang giai đoạn kiểm tra bất động sau va đập.}@*)
    fallState = FALL_IMPACT;
(*@\VNCodeComment{Lưu thời điểm va đập để tính cửa sổ xác nhận tiếp theo.}@*)
    fallStateMs = now;
(*@\VNCodeComment{Hủy chuỗi nếu sau 1.200 ms vẫn không xuất hiện va đập đủ mạnh.}@*)
  } else if (now - fallStateMs > 1200) {
(*@\VNCodeComment{Trở về trạng thái bình thường vì sự kiện rơi tự do không hoàn chỉnh.}@*)
    fallState = FALL_NORMAL;
(*@\VNCodeComment{Xóa mốc gia tốc thấp để lần giám sát kế tiếp bắt đầu sạch.}@*)
    lowGStartMs = 0;
(*@\VNCodeComment{Kết thúc phép kiểm tra va đập sau rơi tự do.}@*)
  }
(*@\VNCodeComment{Thoát khỏi nhánh rơi tự do.}@*)
  break;

(*@\VNCodeComment{Bắt đầu nhánh xử lý sau khi đã phát hiện va đập.}@*)
case FALL_IMPACT:
(*@\VNCodeComment{Yêu cầu đã qua 1.800 ms và mức chuyển động nhỏ hơn 0,08 g.}@*)
  if (now - fallStateMs > 1800 && motionLevel < 0.08f &&
(*@\VNCodeComment{Đồng thời yêu cầu tư thế tĩnh có tổng gia tốc gần 1 g, trong khoảng 0,72--1,28 g.}@*)
      accelerationG > 0.72f && accelerationG < 1.28f) {
(*@\VNCodeComment{Chuyển sang cảnh báo té ngã khi toàn bộ điều kiện bất động đều đúng.}@*)
    fallState = FALL_ALERT;
(*@\VNCodeComment{Lưu thời điểm phát cảnh báo để phục vụ hiển thị, còi và thao tác hủy.}@*)
    fallStateMs = now;
(*@\VNCodeComment{Ghi thông báo cảnh báo ra Serial khi cổng chẩn đoán đang được mở.}@*)
    Serial.println("[ALERT] PHAT HIEN TE NGA - CAN KIEM TRA NGUOI DUNG");
(*@\VNCodeComment{Nếu sau 5 giây vẫn không đủ điều kiện bất động thì xem va đập là sự kiện giả.}@*)
  } else if (now - fallStateMs > 5000) {
(*@\VNCodeComment{Đưa máy trạng thái về bình thường để tiếp tục giám sát.}@*)
    fallState = FALL_NORMAL;
(*@\VNCodeComment{Kết thúc nhánh hết thời hạn chờ sau va đập.}@*)
  }
(*@\VNCodeComment{Thoát khỏi nhánh trạng thái va đập.}@*)
  break;

(*@\VNCodeComment{Bắt đầu nhánh cảnh báo đã được xác nhận.}@*)
case FALL_ALERT:
(*@\VNCodeComment{Giữ nguyên cảnh báo cho đến khi người dùng hủy từ dashboard hoặc khởi động lại.}@*)
  break;
\end{lstlisting}

\subsection{Đếm bước}

Độ lớn gia tốc được tách thành thành phần nền và thành phần chuyển động. Một đỉnh chỉ được xem là bước khi vượt ngưỡng, cách đỉnh trước ít nhất 280 ms và nằm trong chuỗi có chu kỳ không quá 1,8 giây. Bước đầu tiên chỉ được cộng sau khi xuất hiện đỉnh thứ hai, nhờ đó việc cầm thiết bị lên không làm tăng bộ đếm.

\Needspace{0.38\textheight}
\begin{lstlisting}[style=firmware,caption={Lọc gia tốc động và xác nhận chuỗi bước}]
(*@\VNCodeComment{Cập nhật thành phần trọng lực bằng bộ lọc thông thấp với hệ số 0,03.}@*)
gravity += 0.03f * (totalG - gravity);
(*@\VNCodeComment{Loại thành phần trọng lực khỏi tổng gia tốc để thu phần gia tốc do bước chân.}@*)
float dynamicG = totalG - gravity;
(*@\VNCodeComment{Lọc tiếp gia tốc động với hệ số 0,25 để giảm nhiễu cao tần.}@*)
dynamicFiltered += 0.25f * (dynamicG - dynamicFiltered);
(*@\VNCodeComment{Dòng trống phân tách bước lọc tín hiệu với bước phát hiện một chu kỳ bước.}@*)

(*@\VNCodeComment{Mở cờ chờ đỉnh dương sau khi tín hiệu đã đi qua đáy âm nhỏ hơn -0,05 g.}@*)
if (dynamicFiltered < -0.05f) peakArmed = true;
(*@\VNCodeComment{Yêu cầu tín hiệu vượt đỉnh dương 0,11 g khi cờ chờ đã được mở.}@*)
if (peakArmed && dynamicFiltered > 0.11f &&
(*@\VNCodeComment{Giữ khoảng cách tối thiểu 280 ms giữa hai đỉnh và loại xung mạnh từ 1,8 g trở lên.}@*)
    now - lastPeakMs > 280 && totalG < 1.8f) {
(*@\VNCodeComment{Tính khoảng giữa hai đỉnh; đặt bằng 0 nếu chưa từng có đỉnh trước đó.}@*)
  uint32_t interval = lastPeakMs == 0 ? 0 : now - lastPeakMs;
(*@\VNCodeComment{Chấp nhận nhịp bước có chu kỳ từ 280 đến 1.800 ms.}@*)
  if (interval >= 280 && interval <= 1800) {
(*@\VNCodeComment{Kiểm tra chuỗi đi bộ đã được xác nhận từ một cặp đỉnh trước hay chưa.}@*)
    if (walkingSequence) {
(*@\VNCodeComment{Khi chuỗi đã hợp lệ, mỗi đỉnh mới làm tăng bộ đếm thêm một bước.}@*)
      stepCount++;
(*@\VNCodeComment{Đi vào nhánh khởi tạo khi vừa thu được cặp đỉnh hợp lệ đầu tiên.}@*)
    } else {
(*@\VNCodeComment{Cộng hai bước cùng lúc để bù cho đỉnh đầu tiên chưa được tính trước khi xác nhận chuỗi.}@*)
      stepCount += 2;
(*@\VNCodeComment{Đánh dấu hệ thống đang trong một chuỗi đi bộ hợp lệ.}@*)
      walkingSequence = true;
(*@\VNCodeComment{Kết thúc nhánh lựa chọn cách cộng bước.}@*)
    }
(*@\VNCodeComment{Đi vào nhánh này khi khoảng đỉnh nằm ngoài miền nhịp bước cho phép.}@*)
  } else {
(*@\VNCodeComment{Hủy trạng thái chuỗi đi bộ để lần kế tiếp phải xác nhận lại.}@*)
    walkingSequence = false;
(*@\VNCodeComment{Kết thúc kiểm tra khoảng thời gian giữa hai đỉnh.}@*)
  }
(*@\VNCodeComment{Lưu mốc đỉnh hiện tại cho phép tính khoảng đỉnh ở lần sau.}@*)
  lastPeakMs = now;
(*@\VNCodeComment{Đóng cờ chờ đỉnh, buộc tín hiệu phải qua một đáy âm mới trước bước tiếp theo.}@*)
  peakArmed = false;
(*@\VNCodeComment{Kết thúc điều kiện xác nhận đỉnh dương.}@*)
}
\end{lstlisting}

\subsection{UART và thu dữ liệu}

Việc đọc UART nằm trong mọi vòng lặp để không tràn bộ đệm. Phần thu dòng, checksum và parser được viết trực tiếp, không dùng TinyGPS++ hoặc thư viện GPS bên ngoài.

\subsection{Kiểm tra và giải mã NMEA}

Em gom dữ liệu từ ký tự ``\$'' đến cuối dòng, tính XOR phần thân và so sánh với checksum sau dấu ``*''. Chỉ câu đúng checksum mới được tách trường. Câu GGA cung cấp chất lượng fix, số vệ tinh và vị trí; câu RMC cho biết bản tin vị trí có hợp lệ hay không.

Vĩ độ/kinh độ từ dạng độ--phút được đổi sang độ thập phân:

\[
 d_{decimal}=d+\frac{m}{60},
\]

và đổi dấu cho bán cầu Nam hoặc Tây. Vị trí chỉ được đánh dấu FIX khi câu hợp lệ báo chất lượng tốt, ký hiệu bán cầu đúng và tọa độ nằm trong miền cho phép. Khi bản tin mới báo mất fix, trạng thái chuyển sang NOFIX; tọa độ hợp lệ gần nhất vẫn được giữ riêng để dùng trong màn hình cảnh báo.

\Needspace{0.34\textheight}
\begin{lstlisting}[style=firmware,caption={Tính và kiểm tra checksum NMEA trước khi tách trường}]
(*@\VNCodeComment{Khai báo hàm kiểm tra checksum và nhận chuỗi NMEA dạng ký tự làm đầu vào.}@*)
bool NMEA_ChecksumOK(const char *sentence) {
(*@\VNCodeComment{Loại con trỏ rỗng hoặc chuỗi không bắt đầu bằng ký tự đô-la theo chuẩn NMEA.}@*)
  if (!sentence || sentence[0] != '$') return false;
(*@\VNCodeComment{Tìm dấu sao, vị trí phân cách phần dữ liệu với hai chữ số checksum.}@*)
  const char *star = strchr(sentence, '*');
(*@\VNCodeComment{Loại câu không có dấu sao hoặc không đủ hai ký tự checksum phía sau.}@*)
  if (!star || strlen(star) < 3) return false;
(*@\VNCodeComment{Đổi chữ số hexadecimal thứ nhất sau dấu sao thành bốn bit cao.}@*)
  int8_t high = NMEA_HexNibble(star[1]);
(*@\VNCodeComment{Đổi chữ số hexadecimal thứ hai thành bốn bit thấp.}@*)
  int8_t low = NMEA_HexNibble(star[2]);
(*@\VNCodeComment{Loại câu có một trong hai chữ số checksum không hợp lệ.}@*)
  if (high < 0 || low < 0) return false;
(*@\VNCodeComment{Dòng trống phân tách bước đọc checksum nhận được với bước tự tính checksum.}@*)

(*@\VNCodeComment{Khởi tạo biến checksum tính toán bằng 0.}@*)
  uint8_t checksum = 0;
(*@\VNCodeComment{Duyệt các ký tự sau dấu đô-la và dừng ngay trước dấu sao.}@*)
  for (const char *p = sentence + 1; p < star; p++) {
(*@\VNCodeComment{XOR từng byte vào biến tích lũy đúng theo quy tắc checksum NMEA.}@*)
    checksum ^= (uint8_t)*p;
(*@\VNCodeComment{Kết thúc vòng lặp sau khi đã xử lý toàn bộ phần thân câu.}@*)
  }
(*@\VNCodeComment{Ghép bốn bit cao và bốn bit thấp thành byte checksum nhận được.}@*)
  uint8_t expected = (uint8_t)((high << 4) | low);
(*@\VNCodeComment{Chỉ chấp nhận câu khi checksum tự tính bằng checksum đi kèm.}@*)
  return checksum == expected;
(*@\VNCodeComment{Kết thúc thân hàm.}@*)
}
\end{lstlisting}

\subsection{Trạng thái vận hành}

\begin{itemize}
  \item \textbf{WAIT:} chưa nhận byte GPS;
  \item \textbf{LOST:} từng nhận nhưng mất dữ liệu quá 3 giây;
  \item \textbf{NOFIX:} đang nhận NMEA nhưng chưa có vị trí;
  \item \textbf{FIX:} có tọa độ hợp lệ gần thời điểm hiện tại.
\end{itemize}

\subsection{Dữ liệu IMU/GPS và thao tác trên dashboard}

Nhóm trường do em phụ trách được chia rõ theo chức năng:
\begin{itemize}
  \item chuyển động: \Code{steps}, \Code{fall}, \Code{acceleration}, \Code{motion};
  \item trạng thái GPS: \Code{gpsState}, \Code{satellites}, \Code{gpsBytes};
  \item vị trí: \Code{positionKnown}, \Code{latitude}, \Code{longitude}.
\end{itemize}
Giao diện chỉ tạo liên kết bản đồ khi đã có vị trí; trạng thái FIX hiện tại vẫn được tách khỏi tọa độ hợp lệ gần nhất.

Hai yêu cầu POST được kiểm tra theo đúng biến trạng thái của thuật toán. Đường dẫn \path{/api/reset-steps} đưa bộ đếm bước về 0; \path{/api/reset-alert} đưa máy trạng thái về NORMAL và tắt buzzer ở lần cập nhật kế tiếp. Các thao tác này chỉ hoạt động trong mạng cục bộ do ESP32-S3 phát.

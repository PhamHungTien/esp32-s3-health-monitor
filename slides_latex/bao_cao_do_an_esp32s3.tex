\begin{frame}{PPG: kết quả sớm, xác nhận bằng khoảng RR}
  \centering
  \begin{tikzpicture}[node distance=4mm, b/.style={draw=Ink,rounded corners=1mm,minimum width=25mm,minimum height=10mm,align=center,font=\small}, a/.style={-{Latex[length=2mm]},thick}]
    \node[b] (a) {RED/IR\\100 Hz}; \node[b,right=of a] (b) {Tách DC/AC\\lọc IIR};
    \node[b,right=of b] (c) {Ngưỡng\\thích nghi}; \node[b,right=of c] (d) {Nửa chu kỳ\\$\rightarrow$ sơ bộ};
    \node[b,right=of d,fill=Ink,text=white] (e) {RR đầy đủ\\$\rightarrow$ xác nhận};
    \draw[a](a)--(b);\draw[a](b)--(c);\draw[a](c)--(d);\draw[a](d)--(e);
  \end{tikzpicture}
  \vspace{5mm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\large\bfseries Nhịp tim}\par\vspace{2mm}
      \small BPM $=60.000/\Delta t_{RR}$; miền 40--180 BPM. Phiên đo mới tự làm sạch FIFO và phục hồi nếu bộ dò bị kẹt.
    \end{column}
    \begin{column}{0.48\textwidth}
      {\large\bfseries SpO$_2$}\par\vspace{2mm}
      \small Ratio-of-ratios từ RMS(AC)/DC của RED và IR trong cửa sổ 64 mẫu; kết quả chỉ mang tính nguyên mẫu.
    \end{column}
  \end{columns}
  \vspace{5mm}\par
  \bigmetric{$<1$ s}{BPM sơ bộ}\hfill
  \bigmetric{$\leq1{,}75$ s}{BPM xác nhận}\hfill
  \bigmetric{$\approx0{,}76$ s}{SpO$_2$ sớm nhất}
\end{frame}

\begin{frame}{IMU và GPS dùng chuỗi điều kiện lọc nhiễu}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      {\large\bfseries IMU: bước và té ngã}\par\vspace{2mm}
      \small
      \begin{itemize}\setlength\itemsep{2mm}
        \item Bước: đáy $\rightarrow$ đỉnh, chu kỳ 280--1.800 ms
        \item Hai đỉnh liên tiếp mới xác nhận đi bộ
        \item Té ngã: FREEFALL $\rightarrow$ IMPACT $\rightarrow$ bất động
        \item Thiếu điều kiện: trở về NORMAL
      \end{itemize}
    \end{column}
    \begin{column}{0.47\textwidth}
      {\large\bfseries GPS: chỉ giữ dữ liệu hợp lệ}\par\vspace{2mm}
      \small
      \begin{itemize}\setlength\itemsep{2mm}
        \item Tự gom câu GGA/RMC và kiểm tra checksum XOR
        \item WAIT: chưa có byte; NOFIX: UART vẫn hoạt động
        \item FIX: đã có tọa độ; LOST: ngừng byte quá 3 giây
        \item Tọa độ hợp lệ gần nhất được giữ riêng
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
